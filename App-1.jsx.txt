import { useState } from "react";
import { C } from "./constants.js";
import AnalyzeScreen from "./screens/AnalyzeScreen.jsx";
import ChatScreen from "./screens/ChatScreen.jsx";
import LawsScreen from "./screens/LawsScreen.jsx";
import SignalsScreen from "./screens/SignalsScreen.jsx";

const TABS = [
  { id: "analyze", icon: "◈", label: "Analyze" },
  { id: "signals", icon: "⚡", label: "Signals" },
  { id: "chat",    icon: "◇", label: "Ask AI"  },
  { id: "laws",    icon: "⬡", label: "Laws"    },
];

export default function App() {
  const [tab, setTab] = useState("analyze");
  const [savedSignals, setSavedSignals] = useState([]);

  const saveSignal = (signal) => {
    setSavedSignals(prev => [signal, ...prev].slice(0, 50));
  };

  return (
    <div style={{ display: "flex", flexDirection: "column", height: "100vh", background: C.bg, overflow: "hidden" }}>

      {/* Status bar spacer */}
      <div style={{ height: "env(safe-area-inset-top, 0px)", background: "#0a0812" }} />

      {/* Header */}
      <div style={{
        background: "linear-gradient(180deg,#0f0c1a 0%,#06050a 100%)",
        borderBottom: `1px solid ${C.border}`,
        padding: "14px 18px 10px",
        display: "flex", alignItems: "center", justifyContent: "space-between",
        flexShrink: 0,
      }}>
        <div>
          <div style={{ fontFamily: "'Cinzel',serif", fontSize: 16, fontWeight: 700, color: C.gold, letterSpacing: 4 }}>
            SHARINGAN ORACLE
          </div>
          <div style={{ fontSize: 8, color: C.muted, letterSpacing: 2, marginTop: 2 }}>
            BANQU ICS · JEAFX FIBONACCI · MT5 READY
          </div>
        </div>
        <div style={{ width: 36, height: 36, borderRadius: "50%", background: `${C.gold}15`,
          border: `1px solid ${C.goldDim}`, display: "flex", alignItems: "center", justifyContent: "center" }}>
          <span style={{ fontFamily: "'Cinzel',serif", fontSize: 16, color: C.gold }}>◈</span>
        </div>
      </div>

      {/* Screen content */}
      <div style={{ flex: 1, overflow: "hidden", position: "relative" }}>
        {tab === "analyze" && <AnalyzeScreen onSaveSignal={saveSignal} />}
        {tab === "signals" && <SignalsScreen signals={savedSignals} />}
        {tab === "chat"    && <ChatScreen />}
        {tab === "laws"    && <LawsScreen />}
      </div>

      {/* Bottom navigation */}
      <div style={{
        background: "#0a0812",
        borderTop: `1px solid ${C.border}`,
        display: "flex",
        paddingBottom: "env(safe-area-inset-bottom, 0px)",
        flexShrink: 0,
      }}>
        {TABS.map(t => (
          <button key={t.id} onClick={() => setTab(t.id)} style={{
            flex: 1, display: "flex", flexDirection: "column", alignItems: "center",
            justifyContent: "center", padding: "10px 0", background: "none", border: "none",
            cursor: "pointer", borderTop: `2px solid ${tab === t.id ? C.gold : "transparent"}`,
            transition: "all 0.2s",
          }}>
            <span style={{ fontSize: 16, color: tab === t.id ? C.gold : C.muted, marginBottom: 2 }}>{t.icon}</span>
            <span style={{ fontSize: 9, color: tab === t.id ? C.gold : C.muted,
              fontFamily: "'IBM Plex Mono',monospace", letterSpacing: 1 }}>{t.label}</span>
          </button>
        ))}
      </div>
    </div>
  );
}

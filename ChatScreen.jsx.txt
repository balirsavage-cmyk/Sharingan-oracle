import { useState, useRef, useEffect } from "react";
import { C, IS, SYSTEM_PROMPT } from "../constants.js";

const STARTERS = [
  "Explain the ICS flow",
  "How does Law 01 work?",
  "What is a Gemini candle?",
  "Walk me through Law 09",
  "Explain Move Origin TP",
  "How does Law 11 work?",
  "What is candlestick delegation?",
  "How to identify sheath powers?",
];

export default function ChatScreen() {
  const [msgs, setMsgs]   = useState([]);
  const [input, setInput] = useState("");
  const [loading, setLoading] = useState(false);
  const endRef = useRef(null);

  useEffect(() => { endRef.current?.scrollIntoView({ behavior:"smooth" }); }, [msgs]);

  const send = async (text) => {
    const content = text || input;
    if (!content.trim()) return;
    const userMsg = { role:"user", content };
    const newMsgs = [...msgs, userMsg];
    setMsgs(newMsgs); setInput(""); setLoading(true);
    try {
      const resp = await fetch("https://api.anthropic.com/v1/messages", {
        method:"POST", headers:{"Content-Type":"application/json"},
        body: JSON.stringify({ model:"claude-sonnet-4-20250514", max_tokens:1000, system:SYSTEM_PROMPT, messages:newMsgs })
      });
      const d = await resp.json();
      const reply = d.content?.map(b=>b.text||"").join("")||"Error.";
      setMsgs([...newMsgs, { role:"assistant", content:reply }]);
    } catch { setMsgs([...newMsgs, { role:"assistant", content:"⚠️ Connection error." }]); }
    setLoading(false);
  };

  return (
    <div style={{ display:"flex", flexDirection:"column", height:"100%" }}>

      {/* Messages */}
      <div style={{ flex:1, overflowY:"auto", padding:"14px 14px 8px" }}>
        {msgs.length === 0 && (
          <div style={{ textAlign:"center", paddingTop:32, opacity:0.4 }}>
            <div style={{ fontSize:28, color:C.gold }}>◇</div>
            <div style={{ fontFamily:"'Cinzel',serif", fontSize:13, color:C.gold, letterSpacing:3, marginTop:10 }}>ASK THE ORACLE</div>
            <div style={{ fontSize:10, color:C.muted, marginTop:8, lineHeight:1.9 }}>Ask anything about ICS flow,<br/>Sharingan Laws, or Fibonacci.</div>
            <div style={{ display:"flex", flexWrap:"wrap", gap:6, justifyContent:"center", marginTop:18 }}>
              {STARTERS.map(q=>(
                <button key={q} onClick={()=>send(q)} style={{
                  background:"#0d0b14", border:`1px solid ${C.border}`, borderRadius:16,
                  padding:"7px 12px", color:C.muted, fontSize:10,
                  fontFamily:"'IBM Plex Mono',monospace", cursor:"pointer"
                }}>{q}</button>
              ))}
            </div>
          </div>
        )}

        {msgs.map((m,i)=>(
          <div key={i} style={{ display:"flex", justifyContent:m.role==="user"?"flex-end":"flex-start", marginBottom:10 }}>
            <div style={{
              maxWidth:"88%", padding:"11px 14px",
              borderRadius: m.role==="user" ? "14px 14px 4px 14px" : "14px 14px 14px 4px",
              background: m.role==="user" ? "#15112a" : "#0d0b14",
              border:`1px solid ${m.role==="user"?C.purple+"60":C.border}`,
              fontSize:12, color:C.text, lineHeight:1.8,
              fontFamily:"'IBM Plex Mono',monospace", whiteSpace:"pre-wrap"
            }}>{m.content}</div>
          </div>
        ))}

        {loading && (
          <div style={{ display:"flex", marginBottom:10 }}>
            <div style={{ padding:"11px 14px", background:"#0d0b14", border:`1px solid ${C.border}`,
              borderRadius:"14px 14px 14px 4px", fontSize:12, color:C.muted, fontFamily:"'IBM Plex Mono',monospace" }}>
              ◈ Reading the Sharingan...
            </div>
          </div>
        )}
        <div ref={endRef}/>
      </div>

      {/* Input */}
      <div style={{ padding:"10px 14px", borderTop:`1px solid ${C.border}`, background:"#0a0812",
        paddingBottom:"calc(10px + env(safe-area-inset-bottom,0px))" }}>
        <div style={{ display:"flex", gap:8 }}>
          <input
            value={input} onChange={e=>setInput(e.target.value)}
            onKeyDown={e=>{ if(e.key==="Enter"&&!e.shiftKey){ e.preventDefault(); send(); }}}
            placeholder="Ask about ICS, Laws, Fibonacci..."
            style={{ ...IS, flex:1, fontSize:13, padding:"12px 14px" }}
          />
          <button onClick={()=>send()} disabled={loading||!input.trim()} style={{
            padding:"12px 16px", background:`linear-gradient(135deg,#8a6010,${C.gold})`,
            border:"none", borderRadius:8, color:"#06050a",
            fontFamily:"'Cinzel',serif", fontSize:11, fontWeight:700, cursor:"pointer", flexShrink:0
          }}>SEND</button>
        </div>
      </div>
    </div>
  );
}
